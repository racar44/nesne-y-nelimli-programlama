    CREATE DATABASE BilgiIslemEnvanterDB;

USE BilgiIslemEnvanterDB;
GO

IF OBJECT_ID('BakimKayitlari', 'U') IS NOT NULL DROP TABLE BakimKayitlari;
IF OBJECT_ID('Arizalar', 'U') IS NOT NULL DROP TABLE Arizalar;
IF OBJECT_ID('Cihazlar', 'U') IS NOT NULL DROP TABLE Cihazlar;
IF OBJECT_ID('Teknisyenler', 'U') IS NOT NULL DROP TABLE Teknisyenler;
IF OBJECT_ID('Birimler', 'U') IS NOT NULL DROP TABLE Birimler;
GO

CREATE TABLE Birimler (
    BirimId INT IDENTITY(1,1) PRIMARY KEY,
    BirimAdi NVARCHAR(100) NOT NULL,
    KatBilgisi NVARCHAR(50)
);

CREATE TABLE Teknisyenler (
    TeknisyenId INT IDENTITY(1,1) PRIMARY KEY,
    AdSoyad NVARCHAR(100) NOT NULL,
    Uzmanlik NVARCHAR(100),
    AktifMi BIT NOT NULL DEFAULT 1
);

CREATE TABLE Cihazlar (
    CihazId INT IDENTITY(1,1) PRIMARY KEY,
    SeriNo NVARCHAR(50) NOT NULL UNIQUE,
    Ad NVARCHAR(100) NOT NULL,
    Tur NVARCHAR(50) NOT NULL,
    AlisTarihi DATE NOT NULL,
    BakimPeriyoduAy INT NOT NULL,
    BirimId INT NOT NULL,
    CONSTRAINT FK_Cihazlar_Birimler FOREIGN KEY (BirimId) REFERENCES Birimler(BirimId)
);

CREATE TABLE Arizalar (
    ArizaId INT IDENTITY(1,1) PRIMARY KEY,
    CihazId INT NOT NULL,
    TeknisyenId INT NULL,
    Aciklama NVARCHAR(250) NOT NULL,
    Durum NVARCHAR(30) NOT NULL,
    BildirimTarihi DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT FK_Arizalar_Cihazlar FOREIGN KEY (CihazId) REFERENCES Cihazlar(CihazId),
    CONSTRAINT FK_Arizalar_Teknisyenler FOREIGN KEY (TeknisyenId) REFERENCES Teknisyenler(TeknisyenId)
);

CREATE TABLE BakimKayitlari (
    BakimId INT IDENTITY(1,1) PRIMARY KEY,
    CihazId INT NOT NULL,
    TeknisyenId INT NOT NULL,
    BakimTarihi DATE NOT NULL,
    Notlar NVARCHAR(250),
    CONSTRAINT FK_Bakim_Cihazlar FOREIGN KEY (CihazId) REFERENCES Cihazlar(CihazId),
    CONSTRAINT FK_Bakim_Teknisyenler FOREIGN KEY (TeknisyenId) REFERENCES Teknisyenler(TeknisyenId)
);
